cmake_minimum_required(VERSION 3.18)
project(spatial_libear_oracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED EAR_SOURCE OR NOT EXISTS "${EAR_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "EAR_SOURCE must point to exact pinned libear source checkout")
endif()
set(EAR_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(EAR_EXAMPLES OFF CACHE BOOL "" FORCE)
set(EAR_PACKAGE_AND_INSTALL OFF CACHE BOOL "" FORCE)
set(EAR_USE_INTERNAL_EIGEN OFF CACHE BOOL "" FORCE)
set(EAR_USE_INTERNAL_XSIMD OFF CACHE BOOL "" FORCE)
add_subdirectory("${EAR_SOURCE}" libear-build EXCLUDE_FROM_ALL)
add_executable(libear_oracle gain_oracle.cpp)
target_link_libraries(libear_oracle PRIVATE ear)

add_executable(libear_doe_42 gain_doe.cpp)
target_link_libraries(libear_doe_42 PRIVATE ear)
